import { EmptyState } from "@/components/ui/empty-state";
import { Button } from "@/components/ui/button";
import { FileText, Inbox, Search, ImageOff, Plus } from "lucide-react";

A placeholder component shown when a list, page, or section has no content. Provides visual context with an optional icon, title, description, and action.

## Usage

```tsx
import { EmptyState } from "@/components/ui/empty-state";
import { Inbox } from "lucide-react";

<EmptyState
  icon={<Inbox />}
  title="No messages"
  description="You're all caught up."
/>;
```

## Examples

### Default

A simple empty state with icon, title, and description.

<ExampleContainer>
  <EmptyState

  />
</ExampleContainer>

```tsx
<EmptyState
```

### With Action

Add a button or link to guide the user toward a next step.

<ExampleContainer>
  <EmptyState
    icon={<FileText />}
    title="No documents"
    description="Get started by creating your first document."
    action={
      <Button size="sm" variant="primary">
        <Plus />
        New document
      </Button>
    }
  />
</ExampleContainer>

```tsx
<EmptyState
  icon={<FileText />}
  title="No documents"
  description="Get started by creating your first document."
  action={
    <Button size="sm" variant="primary">
      <Plus />
      New document
    </Button>
  }
/>
```

## Components

### EmptyState

The root container. Centers content with a dashed border.

| Prop          | Type        | Default | Description                                  |
| ------------- | ----------- | ------- | -------------------------------------------- |
| `icon`        | `ReactNode` | —       | Icon displayed in a muted container          |
| `title`       | `string`    | —       | Heading text                                 |
| `description` | `string`    | —       | Secondary text, max-width constrained        |
| `action`      | `ReactNode` | —       | Action slot for buttons or links             |
| `children`    | `ReactNode` | —       | Custom content below description and action  |
| `className`   | `string`    | —       | Additional CSS classes                       |
